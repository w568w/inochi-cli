cmake_minimum_required(VERSION 3.24)
project(inochi VERSION 0.1.0 LANGUAGES CXX)
include(FetchContent)
include(GNUInstallDirs)
find_package(Git REQUIRED)
find_program(DUB dub REQUIRED)
find_program(DC NAMES ldc2 dmd gdc REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
find_package(Freetype REQUIRED)
find_package(OpenGL REQUIRED)

# Creator and SDK share a fixed format/algorithm baseline; only their patches vary.
FetchContent_Declare(creator GIT_REPOSITORY https://github.com/Inochi2D/inochi-creator.git GIT_TAG 371e5f4e21117102dbecc82371bfd9a6e2f5df02 GIT_SUBMODULES "" GIT_CONFIG core.autocrlf=false PATCH_COMMAND ${GIT_EXECUTABLE} reset --hard HEAD COMMAND ${GIT_EXECUTABLE} apply --index ${CMAKE_CURRENT_SOURCE_DIR}/patches/creator.patch)
FetchContent_Declare(sdk GIT_REPOSITORY https://github.com/Inochi2D/inochi2d.git GIT_TAG dda3a8f005f808104caa43183ab8753fb989e30f GIT_SUBMODULES "" GIT_CONFIG core.autocrlf=false PATCH_COMMAND ${GIT_EXECUTABLE} reset --hard HEAD COMMAND ${GIT_EXECUTABLE} apply --index ${CMAKE_CURRENT_SOURCE_DIR}/patches/sdk.patch)
FetchContent_Declare(bindings GIT_REPOSITORY https://github.com/Inochi2D/i2d-imgui.git GIT_TAG c6a78f4a7510fd31a86998b7ceedfc2916ecfae0 GIT_SUBMODULES deps/cimgui SOURCE_SUBDIR source)
FetchContent_Declare(i18n GIT_REPOSITORY https://github.com/KitsunebiGames/i18n.git GIT_TAG 75c57ea0889d459b73765d932aec02f9b3e71b80 GIT_SUBMODULES "" GIT_CONFIG core.autocrlf=false PATCH_COMMAND ${GIT_EXECUTABLE} reset --hard HEAD COMMAND ${GIT_EXECUTABLE} apply --index ${CMAKE_CURRENT_SOURCE_DIR}/patches/i18n.patch)
FetchContent_MakeAvailable(creator sdk bindings i18n)
set(cimgui_SOURCE_DIR "${bindings_SOURCE_DIR}/deps/cimgui")
set(imgui_SOURCE_DIR "${cimgui_SOURCE_DIR}/imgui")

add_library(inochi-native SHARED
    source/compat.cpp ${cimgui_SOURCE_DIR}/cimgui.cpp
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
    ${imgui_SOURCE_DIR}/misc/freetype/imgui_freetype.cpp)
target_compile_features(inochi-native PRIVATE cxx_std_17)
target_include_directories(inochi-native PRIVATE ${cimgui_SOURCE_DIR} ${imgui_SOURCE_DIR})
target_compile_definitions(inochi-native PRIVATE CIMGUI_FREETYPE IMGUI_ENABLE_FREETYPE IMGUI_ENABLE_STB_TRUETYPE "IMGUI_IMPL_API=extern \"C\" $<$<PLATFORM_ID:Windows>:__declspec(dllexport)>")
target_link_libraries(inochi-native PRIVATE SDL2::SDL2 Freetype::Freetype OpenGL::GL "$<$<PLATFORM_ID:Windows>:imm32>")
set_target_properties(inochi-native PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")

# Dub links one native target; CMake owns its system-library dependencies.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(platform_dependencies "\"mir-linux-kernel\": \"1.0.1\", \"dportals\": \"0.1.0\",")
endif()
configure_file(cmake/dub.json.in dub.in.json @ONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/dub.json" INPUT "${CMAKE_CURRENT_BINARY_DIR}/dub.in.json")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/dub.selections.json" INPUT "${CMAKE_CURRENT_SOURCE_DIR}/dub.selections.json")
add_custom_target(inochi ALL COMMAND ${DUB} build --root=${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG> --compiler=${DC} --build=$<IF:$<CONFIG:Debug>,debug,release> DEPENDS inochi-native USES_TERMINAL VERBATIM)
install(TARGETS inochi-native RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/inochi${CMAKE_EXECUTABLE_SUFFIX}" DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE CONTRIBUTORS.md DESTINATION ${CMAKE_INSTALL_DATADIR}/inochi)
foreach(upstream IN ITEMS sdk bindings cimgui imgui i18n)
    file(GLOB licenses "${${upstream}_SOURCE_DIR}/LICENSE*")
    install(FILES ${licenses} DESTINATION ${CMAKE_INSTALL_DATADIR}/inochi/licenses/${upstream})
endforeach()
if(WIN32 OR APPLE)
    set(runtime_dirs "$<TARGET_FILE_DIR:inochi-native>")
    foreach(prefix IN LISTS CMAKE_PREFIX_PATH)
        list(APPEND runtime_dirs "${prefix}/bin" "${prefix}/lib")
    endforeach()
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
    include(InstallRequiredSystemLibraries)
    install(CODE "
        include(BundleUtilities)
        function(gp_item_default_embedded_path_override item path)
            set(\${path} @executable_path PARENT_SCOPE)
        endfunction()
        fixup_bundle(\"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}/inochi${CMAKE_EXECUTABLE_SUFFIX}\" \"\"
            \"${runtime_dirs}\")
    ")
endif()
if(APPLE)
    install(CODE "
        file(GLOB binaries \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}/*.dylib\")
        list(APPEND binaries \"\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}/inochi\")
        foreach(binary IN LISTS binaries)
            execute_process(COMMAND codesign --force --sign - \"\${binary}\" COMMAND_ERROR_IS_FATAL ANY)
        endforeach()
    ")
endif()
if(DEFINED VCPKG_INSTALLED_DIR)
    install(DIRECTORY "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/"
        DESTINATION ${CMAKE_INSTALL_DATADIR}/inochi/licenses FILES_MATCHING PATTERN copyright)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_custom_target(native-actions ALL COMMAND ${DUB} build --root=${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG> --compiler=${DC} --build=$<IF:$<CONFIG:Debug>,debug,release> --config=native-actions DEPENDS inochi USES_TERMINAL VERBATIM)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py ${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/inochi${CMAKE_EXECUTABLE_SUFFIX} ${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/inochi-native-actions${CMAKE_EXECUTABLE_SUFFIX})
endif()
set(CPACK_PACKAGE_CONTACT "Inochi CLI contributors")
set(CPACK_SOURCE_IGNORE_FILES "/[.]git/;/[.]dub/;/build/;/vendor/;/out/;/bin/;/dist/;/tests/results/;__pycache__")
include(CPack)
